SIN = lambda f, k=1: math.sin(f * 2 * math.pi * k)


# --------------------------------------------------------------------------
# Ölçüm yardımcıları. Yeni çizim fonksiyonu DEĞİL: figürü boş yüzeye çizip alfa
# kanalından ölçer (ciz.yeni + ciz.figur + PIL). Amaç: elle yazılmış ölçek ve
# koordinat yerine figürün kendi pikselinden türetmek.
# --------------------------------------------------------------------------
def _opak(img):
    w = img.size[0]
    return {(i % w, i // w) for i, v in enumerate(img.getchannel("A").getdata()) if v > 0}


def _bb(img):
    return img.getchannel("A").getbbox()


_ONBELLEK = {}


def olcek(ad, yakin, cizim, oran=0.32):
    """Figür kutusu kadraj yüksekliğinin >= oran'ı olan en küçük s (0.1 adım).
    Kadrajda görünen ızgara yüksekliği = 960 / yakin."""
    anahtar = (ad, yakin, oran)
    if anahtar in _ONBELLEK:
        return _ONBELLEK[anahtar]
    hedef = oran * 960 / yakin
    s = 2.0
    while s <= 9.0:
        t = yeni(560, 420)
        cizim(t, round(s, 1), duz(417, 560))
        bb = _bb(t)
        if bb and bb[1] <= 1:
            break                                    # tuvale sığmıyor
        if bb and bb[3] - bb[1] >= hedef:
            _ONBELLEK[anahtar] = round(s, 1)
            return _ONBELLEK[anahtar]
        s += 0.1
    raise RuntimeError(f"olcek({ad}, yakin={yakin}): %{int(oran * 100)} boya ulaşılamadı")


def s_kisi(yakin, oran=0.32):
    return olcek("kisi", yakin, lambda t, s, z: figur(
        t, kisi, 280, z, s, govde="#7a1e22", bas="sarik", bas_renk="#e8e0d0",
        sakal="#2a1a12", yon=1), oran)


def s_oturan(yakin, oran=0.30):
    return olcek("oturan", yakin, lambda t, s, z: figur(
        t, oturan, 280, z, s, govde="#2a6a3a", bas="sarik", bas_renk="#f4f0e6",
        sakal="#2a1a12", yon=-1, minder="#c8a040"), oran)


def s_atli(yakin):
    """İnsan/at oranı: atlı 2.9 ↔ kisi 2.6 (SKILL.md)."""
    return round(s_kisi(yakin) * 2.9 / 2.6, 1)


def s_muhafiz(yakin, oran=0.32):
    return olcek("muhafiz", yakin, lambda t, s, z: figur(
        t, muhafiz, 280, z, s, faz=0.0, mizrak=True, mesale=True, deel="#2a2a3a"), oran)


def s_gemi(yakin, oran=0.30):
    return olcek("gemi", yakin, lambda t, s, z: figur(t, gemi, 280, 400, s, 1, 0.0), oran)


def _gemi_yarim(s):
    t = yeni(560, 420)
    figur(t, gemi, 280, 400, s, 1, 0.0)
    a, _, b, _ = _bb(t)
    return max(280 - a, b - 280)


def _gemi_denetimi(s, su, w, h):
    """Gövde su çizgisine ulaşmalı (havada gemi olmaz)."""
    for f in (0.0, 0.25, 0.5, 0.75):
        g = yeni(w, h)
        figur(g, gemi, w // 2, su, s, 1, f)
        if _bb(g)[3] < su:
            raise RuntimeError(f"Gemi suya ulaşmıyor: alt={_bb(g)[3]}, su_y={su}")


def _kz(yakin, taban=634, h=300):
    """Kandil zinciri: sprite'ın üst kenarından (tavan) kadrajın üst kenarının biraz altına.
    Kadrajın üst satırı = (taban-12)·(1 - 1/yakin); varsayılan odakY = taban-12."""
    ust_gorunen = (taban - 12) * (1 - 1 / yakin)
    return max(40, int(ust_gorunen - (taban - h)) + 10)


def _kadraj(no, kare, w, h, yakin, esik):
    """Görünür aralık yerel x = w/2 ± 270/yakin. Opak piksellerin en az `esik` oranı içeride
    olmalı; figür sprite'ın üstüne değmemeli (kesik baş = hata)."""
    yarim = 270 / yakin
    a, b = max(int(w / 2 - yarim), 0), min(int(w / 2 + yarim), w)
    for i in (0, 12):
        t = yeni(w, h)
        kare(t, duz(h - 1, w), i)
        alfa = t.getchannel("A")
        toplam = sum(alfa.histogram()[1:])
        bb = alfa.getbbox()
        if not toplam or bb[1] <= 1:
            raise RuntimeError(f"Sahne {no}: sprite yüksekliği (h={h}) yetmiyor ya da kare boş")
        ici = sum(alfa.crop((a, 0, b, h)).histogram()[1:])
        if ici / toplam < esik:
            raise RuntimeError(f"Sahne {no}: figürlerin %{int(100 * ici / toplam)}'i görünür "
                               f"aralıkta (x {a}-{b}); konumları merkeze çek")

TW, TH, CEM_X = 400, 300, 255

_TIRAS = {}


def _cem_yuz(z, s):
    c = yeni(TW, TH)
    figur(c, oturan, CEM_X, z, s, yon=-1, **CEM)
    k = _opak(c)
    ust = min(y for _, y in k)
    alt = max(y for _, y in k)
    return min(x for x, y in k if y <= ust + 0.45 * (alt - ust)), ust, alt


def _berber_el(z, s, bx, kol):
    """El ucu: aynı poz kol=0 ve kol ile çizilir; farkın sağa en uzak noktası el ucudur."""
    a = yeni(TW, TH)
    figur(a, kisi, bx, z, s, yon=1, kol=0.0, **BERBER)
    b = yeni(TW, TH)
    figur(b, kisi, bx, z, s, yon=1, kol=kol, **BERBER)
    fark = _opak(b) - _opak(a)
    return max(fark, key=lambda p: (p[0], -p[1])) if fark else None


def _tiras_ayar(yakin):
    """Ölçek ve berber konumu ölçülerek bulunur (elle BERBER_X yok)."""
    if yakin in _TIRAS:
        return _TIRAS[yakin]
    s = s_kisi(yakin)
    z = duz(TH - 1, TW)
    boy = max(8, round(2.7 * s))                      # bıçak uzunluğu
    kenar, ust, alt = _cem_yuz(z, s)
    for d in range(int(8 * s), int(45 * s)):
        bx = CEM_X - d
        tamam = True
        for kol in (0.45, 0.55, 0.65):                # kol aralığının uçları ve ortası
            el = _berber_el(z, s, bx, kol)
            if el is None or not (-4 <= kenar - (el[0] + boy + 1) <= 8
                                  and ust <= el[1] <= ust + 0.55 * (alt - ust)):
                tamam = False
                break
        if tamam:
            _TIRAS[yakin] = (s, bx, boy, kenar, ust, alt)
            return _TIRAS[yakin]
    raise RuntimeError("Ustura yüze hizalanamadı: kol aralığını daralt ya da CEM_X'i değiştir")


def _ustura(t, el, i, nabiz, boy):

    for k in range(-3, 2):                                # sap: elin içinde

        piksel(t, ex + k, yy - 1, "#f4f8ff")              # üst kenar ışığı
        piksel(t, ex + k, yy, "#dfe6ee")
        piksel(t, ex + k, yy + 1, "#aeb8c6")              # alt kenar gölgesi
    if i % (KARE // nabiz) < 3:                           # nötr beyaz parıltı
        piksel(t, ex + boy + 2, ey - 4, "#ffffff")
        piksel(t, ex + boy + 1, ey - 4, "#cfd8e4")
        piksel(t, ex + boy + 2, ey - 5, "#cfd8e4")

                          and ust <= el[1] <= ust + 0.55 * (alt - ust)):

    _ustura(t, el, i, nabiz, boy)

## Doğrulama (kod tarafının koşturacağı; ben çalıştıramadım)

```sh
python3 -I video/piksel/cem_sultan.py 90 92 105      # ustura, gemi ve olcek denetimleri
python3 -I video/piksel/cem_sultan.py                # tüm sahneler (kadraj + taban_denetimi)
npx remotion still CemSultan --frame=N out.png       # kapak, 90, 92, 93, 94, 110 kareleri
uv run gorsel_kontrol teslimler/<teslim>.md <kare-klasörü>
```

Geçme ölçütleri:
1. **`olcek()`:** Hata verirse mesaj hangi figürün ve hangi `yakin` değerinin %30'a ulaşamadığını yazar. Çare: `yakin` değerini düşür ya da sprite `h`'sini büyüt.
2. **Ustura (90, 105):** `_tiras_ayar` bulup `_tiras_kare` her karede hizayı doğrulamalı. Hata verirse kol aralığını (0,45–0,65) daralt ya da `CEM_X`'i değiştir. El ucu kol farkından ölçüldüğü için kolun öne uzanmadığı bir pozda yanlış nokta verebilir. Bu durumda kare görseline bakılır.
3. **Kadraj (`_kadraj`):** Hata verirse hangi sahnede figürlerin görünür aralıkta olmadığını yazar. Çare: yerel x'leri merkeze çek. 92 ve 107'de tam genişlikte su bandı oranı büyüttüğü için eşik 0,3–0,4'e indirildi.
4. **Kandil formülü:** Kadrajın üst satırının `(taban−12)·(1−1/yakin)` olduğunu varsaydım (yakınlaştırma odağı `odakY` varsayılanı `taban−12`). Render'da lamba kadraja girmiyor ya da yazı bandını çok kapatıyorsa `_kz` içindeki `+10` değerini ayarla.
5. **Sahne 92/107 su dikişi:** Sprite'taki 14 satırlık su bandının rengi orta plandaki suyla aynı bantta durmalı. Render'da dikdörtgen kenar görünürse `su_ve_kiyi` renkleri eşitlenir.
6. **Sahne 101–102:** Karelerde hasta, hekim, testi ya da Napoli odası görünmemeli.
7. **Süre:** Seslendirme gelince `hizala_seri.py` çalışır. Toplam 90 sn altında kalmalı (tahmin 82,8 sn).

Kalite kontrole yeniden gönderilebilir. Tek açık kalan madde, cepheden bakan Cem figürü. Bu madde yukarıdaki "yeni figür isteği" tablosunun ilk satırında.

## 0. Yöntem ve güven notu (önce bunu okuyun)

- Bu oturumda kaynak metinleri açıp sayfa numarası doğrulayamadım. Rapordaki bilgiler, bildiğim ve gerçekten var olan eserlere dayanır. Sayfa numarası vermedim, çünkü uydurma sayfa göstermem. Eser adı, yazar ve dönem verdim.
- Her satırda iki bağımsız kaynak gösterebildiysem ana bölümlere koydum. İki kaynağı adlandıramadığım bilgiler ya "tek kaynak" diye işaretlendi ya da **Bölüm 9 (KAYNAKSIZ)** listesine girdi. Bölüm 9'daki hiçbir şey metne girmez.
- **Sahibin ekseniyle çakışma:** brifteki iki nokta kaynak kuralıma takıldı.
  1. **Ustura/berber ayrıntısı.** Dönem kaynağında ya da iki bağımsız kaynakta bulamadım. Bu yüzden KAYNAKSIZ listesine aldım.
  2. **"Usturayı yalnızca halk anlatısı söyler" cümlesi.** "Yalnızca" diyebilmek için tüm kaynakları taramış olmam gerekir, ben tarayamadım. Cümleyi kanıtlayamam.

  **Kararım:** ustura, olgu olarak değil, en fazla "rivayet" olarak anılabilir. Rivayetin varlığını da iki kaynakla gösteremediğim için varsayılan öneri, tıraşı ana olgu yapmayan kanca ve kapanıştır (Bölüm 10). Sahibin ekseni korunursa risk Bölüm 10'daki kutuda yazılıdır. Gerekçe: ajansın kuralı "kaynaksız cümle metne girmez" diyor ve tıraşın dönem belgesi yok.
- Briftekiyle uyumlu olduğunu not ettiğim noktalar: Burchard'ın ölümde zehirden söz etmemesi, Guicciardini'nin zehir demesi, İnalcık'ın hastalık yorumu. Vatin'in hastalık dediğini brif söylüyor. Ben o eserde ilgili sayfayı açıp teyit edemedim. Metinde güvenle "İnalcık gibi tarihçiler" denebilir. Vatin'in adı metne gerekmez.

## 1. Konu ve gerekçe

İş konusunu verdi: **Cem Sultan.** Fatih'in oğlu ve II. Bayezid'in kardeşidir. Kaybettiği taht savaşından sonra yaklaşık 13 yıl Avrupa'da elden ele dolaştı. 1495'te Napoli'de öldü.

Sahibin kararıyla iki hikâye birleşiyor:
- **Belge:** 31 Mayıs 1492'de Kutsal Mızrak'ın Roma'ya gelişi.
- **Rivayet:** ölümün zehir şüphesi.

**Metin için işe yarayan asıl çatışma şu:** Bayezid, kardeşini kurtarmak yerine onu elinde tutan Papa'ya, İsa'yı deldiğine inanılan bir mızrak yolluyor. Bu çatışma mızrak hediyesinden çıkıyor, tıraş sahnesine ihtiyaç duymuyor. Tıraşı ana ağırlık yapmak, kaynak açısından en zayıf noktaya yaslanmak demektir. Aşağıda bunun yerine sağlam bir "açık döngü" öneriyorum.

## 7. Tartışmalı noktalar

### T1. Cem'in ölüm nedeni
| Kaynak | Ne diyor |
|---|---|
| **Guicciardini** (*Storia d'Italia*, yaklaşık 1537-40, Kitap I). Geç dönem anlatısı, Borgia karşıtı bir yazar. | Cem'in VI. Alexander'ın verdiği zehirle öldüğünü söyler. |
| **Burchard** (*Liber notarum*, Roma). Papalık törenlerinin tanığı. | Ölümü haber olarak kaydeder, zehirden söz etmez. Napoli'de olmadığı için sessizliği kanıt sayılmaz. |
| **İnalcık** (EI², "Djem") | Doğal nedenleri öne çıkarır. |
| **Vatin** (brife göre) | Hastalık der. Ben sayfayı açıp teyit edemedim. |

**Sonuç:** Belge zehiri kanıtlamıyor. Zehir iddiası Guicciardini ile bilinir. Modern tarihçiler hastalığı öne çıkarır.

### T2. Ölüm günü
24 mi 25 Şubat mı? Kaynaklar ayrışıyor. Metinde "Şubat 1495" yazılsın.

### T3. Ölüm yeri
Napoli (Castel Capuano) en yaygın aktarım. Bazı anlatılarda hastalık Capua'da başlar. Metinde "Napoli" yeterli.

### T4. Mızrağın gerçekliği
- Kutsal Mızrak'ın üç ayrı sahibi vardır: Roma, Viyana ve Eçmiyazin (S11 Catholic Encyclopedia, S12 Britannica). Hiçbirinin gerçekliği kesin kanıtlanmış değildir.
- Antakya'da 1098'de mızrak bulunduğunda bile çağdaşlar tartışmıştır (B8, Raymond d'Aguilers).
- **Metinde:** "İsa'yı deldiğine inanılan mızrak." Mızrağın gerçek olduğu iddia edilmez.

### T5. Hediyenin amacı
Brifteki "tutsak kalsın diye" bir yorumdur. Belgeli olan, mızrağın Cem anlaşması bağlamında Roma'ya gittiğidir (S2, S6). Niyet olarak yazmak için ikinci bir kaynak yok. "Kardeşi hâlâ Papa'nın elindeyken" denebilir.

### T6. Tutsaklık süresi
13 yıl mı 12,5 yıl mı? Bölüm 5'teki hesaba bakın. "13 yıla yakın" denir.

### T7. "Saltanatta kardeşlik yoktur" sözü
Anlatım gelenekte vardır ama birincil kaynakta tam sözcüğünü teyit edemedim. Metne alınacaksa "bir anlatıma göre" denmeli, olgu gibi yazılmamalı.

## 9. KAYNAKSIZ: metne girmesin

Bu listedeki hiçbir bilgi metne girmez.

1. **Zehirli ustura / berber ayrıntısı.** Dönem kaynağını ve iki bağımsız kaynağı gösteremedim. (Arama izi, kanıt değil: Türkçe popüler tarih anlatıları.)
2. **Yıllık ödeme tutarı** (Rodos antlaşması, papalık ödemeleri). Rakamlar kaynaklarda ve literatürde farklı aktarılıyor, güvenle yazabileceğim iki kaynak yok.
3. **"300 bin duka" rüşvet iddiası.** Mektubun gerçekliği tartışmalı, rakam tek kaynaklı. Sahibin zaten metne almama kararı var.
4. **Cem'in Fransa'da Philippine-Hélène de Sassenage ile aşkı.** Halk anlatısı gibi duruyor, iki kaynakla doğrulayamadım.
5. **Cem'in doğum günü (22 Aralık).** Yıl iki kaynakla sağlam, gün düzeyi değil.
6. **Cem'in Roma'daki ilk kabul töreninin ayrıntıları** (diz çökmesi, ayak öpmesi). Burchard'da ne yazdığını teyit edemedim.
7. **Mızrağı getiren Osmanlı elçisinin adı.** Doğrulayamadım.
8. **Pollaiuolo'nun yaptığı VIII. Innocentius mezarında Kutsal Mızrak'ın tasviri.** Mezar var, ama iki bağımsız kaynak adını bu oturumda gösteremedim. Güzel bir kanca olurdu, ancak kaynak bulunana kadar kullanılmaz.
9. **Cem'in şairliği.** Tek kaynak (S1). Metinde yalnızca "şair" diye geçecekse flaglenmeli, ayrıntı verilmemeli.
10. **Fatih'in ölümü için zehir rivayeti** (Bayezid'e atfedilen). Bu işin dışında, kaynağı da bulunmadı.
11. **Naaşın teslim bedeli ve pazarlık ayrıntıları.** Gaeta'daki bekleyiş ve Bursa'ya taşınma tarihi yaklaşık, ayrıntı yok.
12. **Guicciardini'nin zehir iddiasında tam olarak ne verildiği** (şeker, içki, toz vb.). Metinden teyit edemedim.

## 10. Kanca adayı olgular ve gönderi fikirleri

### Metne girebilecek, kaynaklı olgular

1. **Cem, 12 yıl 7 ay boyunca elden ele geçti.** Rodos, Fransa'nın komanderileri, Roma, Fransız ordusu. (S3, S2, S4)
2. **31 Mayıs 1492'de Roma'ya "İsa'yı deldiğine inanılan mızrak" geldi.** Gönderen, kardeşinin tutsaklığını elinde tutan Papa'ya hediye eden Bayezid. (B1 Burchard, S2, S6)
3. **Bayezid, Cem'in gözetimi için Şövalyelere ödeme yaptı.** Tutar metinde yok. (S2, S3)
4. **Cem Roma'ya devredilen yıl (1489) Aubusson kardinal oldu.** Bağ kurulurken "aynı yıl" denmeli, "karşılığında" denmemeli. (S3, S6)
5. **Cem, Charles'a Ocak 1495'te teslim edildi, tam dört hafta sonra öldü.** (B2, S6, S7)
6. **Zehir iddiasının tek kaynağı olarak bilinen anlatı Guicciardini'dir.** Burchard sessiz, İnalcık hastalık der. (B2, B1, S1)
7. **Mızrağın üç sahibi var: Roma, Viyana, Eçmiyazin.** (S11, S12)
8. **Cem'in naaşı, ölümünden yaklaşık dört yıl sonra Bursa'ya geldi.** Yıl yaklaşık verilmeli. (S9, S3)

### Gönderi için 3 fikir

**A. "O mızrak neden Roma'ya gitti?" (önerdiğim varsayılan eksen)**
- *Kanca:* "Bir sultan, kardeşini kurtarmak yerine Papa'ya İsa'nın mızrağı olduğuna inanılan bir hediye yolluyor."
- Yapı: kanca → 13 yıla yakın tutsaklık (Rodos, Fransa, Roma) → ama: mızrak → son: Napoli'de 1495. Zehir mi, hastalık mı? Guicciardini zehir der, Burchard sessiz, İnalcık hastalık der. İzleyiciye soru.
- Kaynak riski sıfıra yakın.

**B. Sahibin ekseni: "Cem Sultan'ın Zehirli Tıraşı" (koşullu)**
- Tıraş ve ustura, Bölüm 9'da. Kullanılacaksa:
  1. Sadece "rivayete göre" etiketiyle, bir tek cümle.
  2. Ekranda ve seslendirmede "ustura için belge yok" ifadesi yer almalı.

**C. "Dört haftalık tutsaklık"**
- *Kanca:* "Cem Sultan, Papa'nın elinden çıktı. Dört hafta sonra öldü."
- Teslim (Ocak 1495) ile ölüm (Şubat 1495) arasındaki 28 gün. Sonunda zehir mi, hastalık mı sorusu.
- Hem sağlam kaynaklı (B2, S6, S7) hem de tıraşsız bir eksen.

### Kalite kontrol için
- Kaynak riski taşıyan ifadeler: "yalnızca halk anlatısı", "tutsak kalsın diye", "karşılığında kardinal", ustura, para tutarları.
- Mızrak her seferinde "inanılan" ifadesiyle verilmeli.
- Süre "tam 13 yıl" yerine "yaklaşık 13 yıl" olmalı.
